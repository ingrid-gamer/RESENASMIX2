import { useEffect, useState } from 'react'
import { listReviews, removeReview } from '../services/api.js'
import { useAuth } from '../contexts/AuthContext.jsx'
import { Navigate } from 'react-router'
export default function AdminReviews() { const { user, isAdmin } = useAuth(); const [reviews, setReviews] = useState([]); const [error, setError] = useState(''); useEffect(() => { if (!user || !isAdmin) return; listReviews().then(setReviews).catch((err) => setError(err.message)) }, [user, isAdmin]); if (!user) return <Navigate to="/login" replace />; if (!isAdmin) return <Navigate to="/" replace />; async function del(id) { try { await removeReview(id); setReviews((current) => current.filter((review) => review.id !== id)) } catch (err) { setError(err.message) } } return <div className="container"><h1 className="page-title h2">Administrar reseñas</h1><p className="text-muted-soft">Eliminar reseñas permite demostrar la operación Delete del CRUD.</p>{error && <div className="alert alert-danger">{error}</div>}<div className="table-dark-panel table-responsive"><table className="table"><thead><tr><th>ID</th><th>Juego</th><th>Usuario</th><th>Nota</th><th>Comentario</th><th /></tr></thead><tbody>{reviews.map((review) => <tr key={review.id}><td>{review.id}</td><td>{review.videojuego?.titulo}</td><td>{review.usuario?.nombre}</td><td>{review.calificacion}/5</td><td>{review.comentario}</td><td><button className="btn btn-outline-danger btn-sm" onClick={() => del(review.id)}>Eliminar</button></td></tr>)}</tbody></table></div></div> }
